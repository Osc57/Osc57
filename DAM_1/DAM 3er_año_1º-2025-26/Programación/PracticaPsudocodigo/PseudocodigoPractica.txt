    Definir contrasena Como Cadena;
    Definir i, longitud Como Entero;
    Definir mayus, minus, numero, simbolo, espacio Como Entero;

    mayus <- 0;
    minus <- 0;
    numero <- 0;
    simbolo <- 0;
    espacio <- 0;

    Escribir "Introduce una contraseña: ";
    Leer contrasena;

    longitud <- Longitud(contrasena);

    Si longitud < 8 Entonces
        Escribir "Contraseña inválida: demasiado corta";
        FinProceso
    FinSi

    Si EsNumero(Subcadena(contrasena,1,1)) Entonces
        Escribir "Contraseña inválida: no puede empezar por número";
        FinProceso
    FinSi

    Para i <- 1 Hasta longitud Hacer

        Si contrasena[i] = " " Entonces
            espacio <- 1;
        FinSi

        Si contrasena[i] >= "A" Y contrasena[i] <= "Z" Entonces
            mayus <- mayus + 1;
        FinSi

        Si contrasena[i] >= "a" Y contrasena[i] <= "z" Entonces
            minus <- minus + 1;
        FinSi

        Si contrasena[i] >= "0" Y contrasena[i] <= "9" Entonces
            numero <- numero + 1;
        FinSi

        Si contrasena[i] En {"!", "@", "#", "$", "%", "&", "/", "?"} Entonces
            simbolo <- simbolo + 1;
        FinSi

    FinPara

    Si espacio = 1 Entonces
        Escribir "Contraseña inválida: contiene espacios";
        FinProceso
    FinSi

    Si mayus = 0 O minus = 0 O numero = 0 O simbolo = 0 Entonces
        Escribir "Contraseña inválida: no cumple los requisitos";
    Sino
        Escribir "Contraseña válida";
    FinSi

FinProceso
